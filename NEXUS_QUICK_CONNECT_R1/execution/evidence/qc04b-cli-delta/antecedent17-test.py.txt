@pytest.fixture
def quick_cli(terminal, tmp_path, monkeypatch):
    from okto_nexus.adapters.inbound.cli import connect, main, work
    from okto_nexus.adapters.outbound import git_project_hint

    t = terminal
    trusted_git = git_project_hint._trusted_git
    home = tmp_path / 'quick-cli-home'
    lifecycle = []
    service = SimpleNamespace(
        adapter=SimpleNamespace(home=home, host='127.0.0.1', port=8202),
        inspect=lambda: lifecycle.append('inspect') or {'state': 'ready'},
        ensure_running=lambda **_: lifecycle.append('ensure') or {'state': 'ready'})
    monkeypatch.setattr(connect, 'resolve_local_service', lambda _args: service)
    monkeypatch.setattr(connect, 'LocalOperatorHTTP', lambda **_kwargs: t.call)
    monkeypatch.setattr(connect.sys, 'stdin', SimpleNamespace(isatty=lambda: False))
    monkeypatch.chdir(t.selection['project'])

    def forbidden_chat(*_args, **_kwargs):
        pytest.fail('noninteractive CLI entered project work')

    def forbidden_git(*_args, **_kwargs):
        pytest.fail('plan/retained review attempted executable discovery')

    monkeypatch.setattr(work, 'interactive_work', forbidden_chat)
    monkeypatch.setattr(git_project_hint, '_trusted_git', forbidden_git)

    def run(tokens):
        # Explicit own home before terminal_target_selection: never inspect a
        # personal saved operator profile/config context, even with inert service.
        return main.main([*tokens, '--home', str(home)])

    return SimpleNamespace(t=t, home=home, service=service, lifecycle=lifecycle,
                           run=run, project=Path(t.selection['project']), trusted_git=trusted_git)


def invoke(cli, capsys, tokens, expected=0):
    code = cli.run(tokens)
    captured = capsys.readouterr()
    assert code == expected, captured.out + captured.err
    value = json.loads(captured.out)
    assert isinstance(value, dict)
    return value


def first_connection(cli, capsys, *, profile='first', name='Logical App', connect_id=None):
    base = ['connect', 'codex', name, '--profile', profile, '--json']
    if connect_id:
        base += ['--id', connect_id]
    review = invoke(cli, capsys, [*base, '--plan'])
    assert review['state'] == 'review' and review['quick_connect']['status'] == 'planned'
    assert cli.t.turns == cli.t.probes == []
    result = invoke(cli, capsys, [*base, '--approve', review['review_revision']])
    assert result['state'] == 'configured'
    return base, review, result


def test_real_http_plan_apply_and_recovery_preserve_public_id_without_test_or_grant_replay(quick_cli, capsys):
    c = quick_cli
    _, review, result = first_connection(c, capsys)
    summary = result['quick_connect']
    assert summary['connect_id'] == 'codex' and summary['connect_id_status'] == 'bound'
    assert summary['operation'] == 'created' and summary['project_name'] == 'Logical App'
    assert summary['project_name_source'] == 'explicit' and summary['project_root'] == str(c.project)
    assert result['session_state'] == 'not_checked' and result['work_state'] == 'not_submitted'
    assert len(c.t.turns) == len(c.t.probes) == 1
    before = _terminal.snapshot(c.t.deps)
    repeated = invoke(c, capsys, ['connect', '--profile', 'first', '--json'])
    assert repeated['quick_connect']['connect_id'] == 'codex'
    assert repeated['quick_connect']['operation'] == 'reused'
    assert repeated['binding_id'] == result['binding_id']
    assert _terminal.snapshot(c.t.deps) == before and len(c.t.turns) == len(c.t.probes) == 1
    assert review['selection']['project'] == str(c.project)


def test_second_codex_explicit_add_uses_reviewer_id_in_same_workspace(quick_cli, capsys):

    base = ['connect', 'codex', 'Logical App', '--profile', 'second', '--id', 'reviewer',

    assert second['quick_connect']['connect_id'] == 'reviewer'
    assert len(c.t.turns) == 2  # two explicitly approved synthetic connection tests

@pytest.mark.parametrize('lost', ['setup:test', 'setup:finish'])
def test_pending_addition_recovers_same_intent_before_collision_without_git_or_replay(quick_cli, capsys, lost):
    c = quick_cli
    base = ['connect', 'codex', 'App', '--profile', 'pending', '--new-connection', '--id', 'reviewer', '--json']
    review = invoke(c, capsys, [*base, '--plan'])
    c.t.fault.after = lost
    uncertain = invoke(c, capsys, [*base, '--approve', review['review_revision']], expected=3)
    assert uncertain['possible_effect'] is True and uncertain['code'] == 'OUTCOME_UNKNOWN'
    assert uncertain['quick_connect']['status'] == 'unknown'
    before = _terminal.snapshot(c.t.deps)
    assert len(c.t.turns) == len(c.t.probes) == 1
    recovered = invoke(c, capsys, base)
    assert recovered['quick_connect']['connect_id'] == 'reviewer'
    assert recovered['quick_connect']['status'] == 'connected'
    assert len(c.t.turns) == len(c.t.probes) == 1
    after = _terminal.snapshot(c.t.deps)
    if lost == 'setup:finish':
        assert after == before
    assert len(after['runtime_execution_grants']) == len(after['execution_bindings']) == 1


def test_legacy_public_id_adoption_only_writes_metadata_preserves_unknown_fields_without_echo(quick_cli, capsys):
    c = quick_cli
    base = ['connect', '--client', 'codex', '--agent', 'subject', '--profile', 'legacy',
            '--project', str(c.project), '--json']
    review = invoke(c, capsys, [*base, '--plan'])
    legacy = invoke(c, capsys, [*base, '--approve', review['review_revision']])
    assert legacy['state'] == 'configured' and 'quick_connect' not in legacy
    path = c.home / 'terminal' / 'profiles' / 'legacy.json'
    saved = json.loads(path.read_text(encoding='utf-8'))
    saved['legacy_private'] = 'PRIVATE_opaque_NOT_PUBLIC'
    saved['selection']['legacy_extension'] = {'opaque': 'PRIVATE_opaque_NOT_PUBLIC'}
    path.write_text(json.dumps(saved), encoding='utf-8')
    before = _terminal.snapshot(c.t.deps)
    posts_before = sum(method == 'POST' for method, _ in c.t.calls)
    adopt = ['connect', 'codex', '--id', 'legacy-public', '--profile', 'legacy', '--json']
    proposal = invoke(c, capsys, [*adopt, '--plan'])
    assert proposal['kind'] == 'public_id_association'
    assert 'PRIVATE_opaque_NOT_PUBLIC' not in json.dumps(proposal)
    result = invoke(c, capsys, [*adopt, '--approve', proposal['review_revision']])
    assert result['quick_connect']['operation'] == 'associated_existing'
    assert result['quick_connect']['connect_id'] == 'legacy-public'
    assert result['binding_id'] == legacy['binding_id']
    assert 'PRIVATE_opaque_NOT_PUBLIC' not in json.dumps(result)
    assert _terminal.snapshot(c.t.deps) == before
    assert len(c.t.turns) == len(c.t.probes) == 1
    assert sum(method == 'POST' for method, _ in c.t.calls) == posts_before + 1
    on_disk = json.loads(path.read_text(encoding='utf-8'))
    assert on_disk['legacy_private'] == 'PRIVATE_opaque_NOT_PUBLIC'
    assert on_disk['selection']['legacy_extension'] == {'opaque': 'PRIVATE_opaque_NOT_PUBLIC'}


def test_tty_customize_id_creates_new_review_preserves_hint_and_invalidates_original_hash(quick_cli, monkeypatch, capsys):
    from okto_nexus.adapters.inbound.cli import connect
    from okto_nexus.adapters.outbound import git_project_hint
    c = quick_cli
    monkeypatch.setattr(connect.sys, 'stdin', SimpleNamespace(isatty=lambda: True))
    discovery_calls = []
    real_discover = git_project_hint.discover_project_name

    def discover(root, *, plan=False):
        discovery_calls.append((root, plan))
        return real_discover(root, plan=plan)

    # The first invocation reads only this temporary nonrepository directory.
    # A restart changing only the ID must retain that hint without another read.
    monkeypatch.setattr(git_project_hint, '_trusted_git', c.trusted_git)
    monkeypatch.setattr(git_project_hint, 'discover_project_name', discover)
    responses = iter(['customize', '1', 'reviewer', 'y'])
    originals = []

    def answer(prompt):
        value = next(responses)
        if value == 'customize':
            paths = list((c.home / 'terminal' / 'profiles').glob('*.json'))
            assert len(paths) == 1
            originals.append((paths[0], json.loads(paths[0].read_text(encoding='utf-8'))))
            assert c.t.turns == c.t.probes == []
        elif value == '1':
            assert prompt == 'Customize: '
        elif value == 'reviewer':
            assert prompt == 'New ID do Harness: '
        else:
            assert 'Confirm this review' in prompt
            assert c.t.turns == c.t.probes == []
        return value

    monkeypatch.setattr('builtins.input', answer)
    assert c.run(['connect', 'codex', '--no-chat']) == 0
    output = capsys.readouterr().out
    assert 'ID do Harness: reviewer' in output
    assert len(discovery_calls) == 1 and discovery_calls[0][0] == c.project
    assert len(c.t.turns) == len(c.t.probes) == 1
    old_path, old = originals[0]
    replaced = json.loads(old_path.read_text(encoding='utf-8'))
    assert replaced['state'] == 'superseded'
    new_path = old_path.with_name(replaced['superseded_by_profile'] + '.json')
    current = json.loads(new_path.read_text(encoding='utf-8'))
    assert current['state'] == 'done'
    assert current['review_revision'] != old['review_revision']
    assert current['request']['client_intent_id'] != old['request']['client_intent_id']
    assert current['selection']['label'] == old['selection']['label']
    assert current['selection']['public_connection']['connect_id'] == 'reviewer'
    assert current['selection']['quick_connect']['name_discovery'] == {'status': 'saved', 'reason': 'saved_selection'}
    before = _terminal.snapshot(c.t.deps)
    monkeypatch.setattr(connect.sys, 'stdin', SimpleNamespace(isatty=lambda: False))
    refused = invoke(c, capsys, ['connect', '--profile', old_path.stem,
        '--approve', old['review_revision'], '--json'], expected=3)
    assert refused.get('quick_connect', {}).get('status') != 'connected'
    resumed = invoke(c, capsys, ['connect', 'codex', '--json'])
    assert resumed['quick_connect']['connect_id'] == 'reviewer'
    assert resumed['quick_connect']['operation'] == 'reused'
    assert len(c.t.turns) == len(c.t.probes) == 1 and _terminal.snapshot(c.t.deps) == before
